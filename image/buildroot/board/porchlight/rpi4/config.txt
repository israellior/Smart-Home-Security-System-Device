start_file=start4.elf
fixup_file=fixup4.dat

kernel=Image

# To use an external initramfs file
#initramfs rootfs.cpio.gz

# Disable overscan assuming the display supports displaying the full resolution
# If the text shown on the screen disappears off the edge, comment this out
disable_overscan=1

# How much memory in MB to assign to the GPU on Pi models having
# 256, 512 or 1024 MB total memory
gpu_mem_256=100
gpu_mem_512=100
gpu_mem_1024=100

# Enable UART0 for serial console on ttyAMA0
dtoverlay=miniuart-bt

# enable 64bits support
arm_64bit=1

# The Waveshare WM8960 audio HAT: the chime, and later the call's microphone
# and speaker. Copied from porch-2's working Raspberry Pi OS config. I2C is how
# the Pi sets the codec's registers, I2S is how the samples travel, and the
# overlay describes the card so the in-tree drivers bind to it - which makes
# it a card named wm8960soundcard, the name porchlightd.json uses.
dtparam=i2c_arm=on
dtparam=i2s=on
dtoverlay=i2s-mmap
dtoverlay=wm8960-soundcard

# The Camera Module 3 (IMX708). The firmware probes the CSI port and applies
# the sensor's overlay itself, which is how porch-2 found it. vc4-kms-v3d is
# what porch-2 ran too: libcamera allocates its frame buffers from the CMA
# pool, and this overlay is what sizes that pool for a camera.
camera_auto_detect=1
dtoverlay=vc4-kms-v3d
